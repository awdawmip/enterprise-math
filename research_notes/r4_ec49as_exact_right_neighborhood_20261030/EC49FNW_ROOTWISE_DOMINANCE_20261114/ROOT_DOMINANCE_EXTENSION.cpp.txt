int root_delta(const RootData &a, const RootData &b) {
  vector<int> from, to;
  for (int j = 0; j < (int)a.labels.size(); ++j) {
    auto it = lower_bound(b.labels.begin(), b.labels.end(), a.labels[j]);
    if (it != b.labels.end() && *it == a.labels[j]) {
      from.push_back(j);
      to.push_back(int(it-b.labels.begin()));
    }
  }
  const int q = from.size();
  int res = 0;
  for (int mask = 0; mask < (1<<q); ++mask) {
    int x=0, y=0;
    for (int j=0; j<q; ++j) if (mask>>j&1) {
      x |= 1<<from[j];
      y |= 1<<to[j];
    }
    if (a.dp[x] >= INF || b.dp[y] >= INF)
      throw runtime_error("root delta invalid common subset");
    res = max(res, a.dp[x]-b.dp[y]);
  }
  return res;
}

bool source_cut_target_lawful(const Context &t, const NodeCert &n) {

      if (!binary_search(v.begin(), v.end(), l)) return false;
    }
  return true;
}
