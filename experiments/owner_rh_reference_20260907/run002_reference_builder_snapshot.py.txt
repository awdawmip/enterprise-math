"""Strict producer for one eta=9/10 N=8 prime+Cauchy reference matrix.

The output encloses matrix entries; it never certifies or assumes an inertia.
Existing Arb diagonal/prime/pole formulas are reused under a SHA256 binding.
All exported endpoints come from exact binary Arb data, never display strings.
"""

from __future__ import annotations

import argparse
from fractions import Fraction
import hashlib
import importlib.util
import json
from pathlib import Path
import platform
import sys
import time

ROOT = Path(__file__).resolve().parents[2]
SOURCE = ROOT / "scripts/rh_log3_n8_arb_certificate.py"
SOURCE_SHA256 = "4e337855d8605125ded80c029022fdc950d1b92cf78ce772480a7b99b9dad49b"
SCHEMA = "owner_rh_reference_bounds_v1"
COMPONENT_SCHEMA = "owner_rh_reference_components_v1"
ETA = Fraction(9, 10)
EXPORT_BITS = 160
LABEL_ENDPOINTS = {
    "O-": ("log(1/2)", "log(1)"),
    "O+": ("log(1)", "log(2)"),
    "S-": ("log(1/3)", "log(1/2)"),
    "S+": ("log(2)", "log(3)"),
}


class Undetermined(RuntimeError):
    """A resource, backend, or enclosure failure; never a spectral verdict."""


def _rational_text(value):
    value = Fraction(value)
    return f"{value.numerator}/{value.denominator}"


def _binary_fraction(exact_arb):
    mantissa, exponent = exact_arb.man_exp()
    mantissa, exponent = int(mantissa), int(exponent)
    if abs(exponent) > 100000:
        raise Undetermined("export exponent exceeds the declared finite-size limit")
    return (Fraction(mantissa << exponent) if exponent >= 0
            else Fraction(mantissa, 1 << (-exponent)))


def export_ball(ball):
    """Directed binary endpoints, then exact outward rounding to 2^-160.

    This avoids enormous denominators for exponentially tiny diagnostic radii.
    Each endpoint is widened by less than 2^-160, with no decimal conversion.
    """
    if not ball.is_finite():
        raise Undetermined("non-finite real ball cannot be exported")
    def on_grid(endpoint, upper):
        mantissa, exponent = endpoint.man_exp()
        mantissa, shift = int(mantissa), int(exponent) + EXPORT_BITS
        if shift >= 0:
            if shift > 4096:
                raise Undetermined("export magnitude exceeds the declared size limit")
            units = mantissa << shift
        else:
            units = -((-mantissa) >> (-shift)) if upper else mantissa >> (-shift)
        return Fraction(units, 1 << EXPORT_BITS)
    lower = on_grid(ball.lower(), False)
    upper = on_grid(ball.upper(), True)
    if lower > upper:
        raise Undetermined("backend exported reversed endpoints")
    return {"lo": _rational_text(lower), "hi": _rational_text(upper)}


def _load_backend(precision):
    import flint
    from flint import acb, arb, ctx, fmpq
    if flint.__version__ != "0.9.0":
        raise Undetermined("this producer requires python-flint==0.9.0")
    ctx.prec = precision
    for value in (Fraction(0), Fraction(2), Fraction(1, 3), Fraction(-1, 3)):
        enclosure = export_ball(arb(fmpq(value.numerator, value.denominator)))
        if not Fraction(enclosure["lo"]) <= value <= Fraction(enclosure["hi"]):
            raise Undetermined("exact endpoint self-check failed")
    if hashlib.sha256(SOURCE.read_bytes()).hexdigest() != SOURCE_SHA256:
        raise Undetermined("inherited builder SHA256 does not match the audited source")
    sys.dont_write_bytecode = True
    spec = importlib.util.spec_from_file_location("owner_inherited_rh_arb", SOURCE)
    inherited = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(inherited)
    return flint, arb, acb, fmpq, inherited


def _labels():
    return [dict(branch=branch, k=k, a=LABEL_ENDPOINTS[branch][0],
                 b=LABEL_ENDPOINTS[branch][1])
            for branch in ("O-", "O+", "S-", "S+")
            for k in range(1, 9)]


def _basis_groups(inherited):
    full = inherited.make_basis(8)
    old = [b for b in full if b["label"] in ("O-", "O+")]
    shell = [b for b in full if b["label"] in ("S-", "S+")]
    assert len(old) == len(shell) == 16
    return old, shell, old + shell


def _check_tail_domain(basis, K, P, arb):
    if type(K) is not int or K < 80 or type(P) is not int or P < 6:
        raise Undetermined("need integer K>=80 and integer P>=6")
    c0 = arb(2 * K) + arb("1/2")
    minimum = None
    for bi in basis:
        for bj in basis:
            rho = ((bi["A"].abs_upper() + bj["A"].abs_upper()) / c0) ** 2
            gap = 1 - rho
            if not gap > 0:
                raise Undetermined("product-denominator convergence was not certified")
            lower = _binary_fraction(gap.lower())
            minimum = lower if minimum is None else min(minimum, lower)
    # Product positivity implies the single-denominator inequalities as well.
    return _rational_text(minimum)


def _mirror_upper(matrix):
    n = len(matrix)
    return [[matrix[min(i, j)][max(i, j)] for j in range(n)] for i in range(n)]


def _export_matrix(matrix):
    return [[export_ball(value) for value in row] for row in matrix]


def _import_matrix(bounds, arb, fmpq):
    def entry(value):
        lo, hi = Fraction(value["lo"]), Fraction(value["hi"])
        if lo > hi:
            raise Undetermined("component cache has reversed bounds")
        # union of rigorously converted rational endpoints encloses the interval.
        return arb(fmpq(lo.numerator, lo.denominator)).union(
            arb(fmpq(hi.numerator, hi.denominator)))
    return [[entry(value) for value in row] for row in bounds]


def _write_json(path, payload):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    data = json.dumps(payload, ensure_ascii=False, sort_keys=True, separators=(",", ":")) + "\n"
    path.write_bytes(data.encode("utf-8"))


def _components(old, shell, basis, K, P, precision, inherited, arb, fmpq,
                component_output, reuse_components, diagnostics):
    parameters = {"K": K, "P": P, "precision_bits": precision, "source_sha256": SOURCE_SHA256}
    if reuse_components is not None:
        cached = json.loads(Path(reuse_components).read_text(encoding="utf-8"))
        if (cached.get("schema") != COMPONENT_SCHEMA or cached.get("parameters") != parameters
                or cached.get("labels") != _labels()):
            raise Undetermined("component cache provenance or parameters mismatch")
        diagnostics["component_cache"] = {
            "path": str(reuse_components),
            "sha256": hashlib.sha256(Path(reuse_components).read_bytes()).hexdigest(),
        }
        diagnostics["arch_tail_radii"] = cached["arch_tail_radii"]
        return tuple(_import_matrix(cached[key], arb, fmpq) for key in ("A", "D", "B_prime"))

    matrices, radii = [], []
    for name, group in (("old", old), ("shell", shell)):
        print(f"building {name} 16x16 arch prefix K={K} and accelerated tail P={P}", flush=True)
        prefix = inherited.arch_partial(group, K)
        tail, max_radius = inherited.accelerated_tail(group, K, P)
        poles = inherited.pole_matrix(group)
        matrices.append(inherited.add_matrices(prefix, tail, poles))
        radii.append(export_ball(max_radius))
    print("building finite prime matrix with exact support selectors", flush=True)
    primes = inherited.prime_matrix(basis)
    A, D = matrices
    for i in range(16):
        for j in range(16):
            A[i][j] += primes[i][j]
            D[i][j] += primes[16+i][16+j]
    A, D = _mirror_upper(A), _mirror_upper(D)
    B_prime = [row[16:] for row in primes[:16]]
    diagnostics["arch_tail_radii"] = {"old": radii[0], "shell": radii[1]}
    if component_output is not None:
        cached = dict(schema=COMPONENT_SCHEMA, parameters=parameters, labels=_labels(),
                      A=_export_matrix(A), D=_export_matrix(D), B_prime=_export_matrix(B_prime),
                      arch_tail_radii=diagnostics["arch_tail_radii"])
        _write_json(component_output, cached)
        diagnostics["component_cache"] = {
            "path": str(component_output),
            "sha256": hashlib.sha256(Path(component_output).read_bytes()).hexdigest(),
        }
    return A, D, B_prime


def _panels(cutoff):
    endpoints = [0, 1]
    while endpoints[-1] * 2 < cutoff:
        endpoints.append(endpoints[-1] * 2)
    if endpoints[-1] != cutoff:
        endpoints.append(cutoff)
    return list(zip(endpoints, endpoints[1:]))


def _cauchy_entry(bi, bj, cutoff, inherited, arb, acb, fmpq, limits, counters):
    if bi["bq"] <= bj["aq"]:
        left, right = bi, bj
    elif bj["bq"] <= bi["aq"]:
        left, right = bj, bi
    else:
        raise Undetermined("Cauchy cross must have disjoint support interiors")
    start_evaluations = counters["evaluations"]

    def integrand(c, analytic):
        # The inherited closed form is meromorphic: only rational operations
        # and exp(c*constant). Pole-containing balls become non-finite naturally.
        counters["evaluations"] += 1
        if counters["evaluations"] > limits["total_eval_limit"]:
            raise Undetermined("global integration evaluation limit reached")
        if counters["evaluations"] - start_evaluations > limits["entry_eval_limit"]:
            raise Undetermined("single-entry integration evaluation limit reached")
        return -inherited.ordered_distinct_laplace(left, right, c) / 2

    result = acb(0)
    panel_evidence = []
    for a, b in _panels(cutoff):
        value = acb.integral(
            integrand, a, b, abs_tol=arb((1, -limits["tolerance_bits"])),
            rel_tol=arb((1, -limits["tolerance_bits"])),
            eval_limit=limits["panel_eval_limit"], depth_limit=limits["depth_limit"],
            deg_limit=limits["degree_limit"],
        )
        if not value.is_finite() or not value.imag.contains(0):
            raise Undetermined("finite Cauchy integral returned a non-finite/nonreal enclosure")
        result += value
        counters["panels"] += 1
        panel_evidence.append(export_ball(value.real.rad()))
    finite_real = result.real
    if not finite_real.is_finite():
        raise Undetermined("finite Cauchy sum is non-finite")
    # Uniform analytic bound proved for every old/shell pair in the intake.
    tail = Fraction(10240, cutoff**3)
    complete = finite_real + arb(0, arb(fmpq(tail.numerator, tail.denominator)).upper())
    return complete, {
        "evaluations": counters["evaluations"] - start_evaluations,
        "finite_part": export_ball(finite_real),
        "finite_radius": export_ball(finite_real.rad()),
        "tail_radius": _rational_text(tail),
        "left_branch": left["label"], "right_branch": right["label"],
        "panel_radius_bounds": panel_evidence,
    }


def build_reference_bounds(*, K=256, P=10, precision=384, cauchy_cutoff=20000,
                           tolerance_bits=60, panel_eval_limit=4000,
                           entry_eval_limit=100000, total_eval_limit=2000000,
                           depth_limit=20, degree_limit=32, component_output=None,
                           reuse_components=None, progress_output=None):
    """Return a JSON-safe ENCLOSED/UNDETERMINED payload; no inertia verdict."""
    started = time.perf_counter()
    parameters = dict(K=K, P=P, precision_bits=precision, cauchy_cutoff=cauchy_cutoff,
                      export_grid_bits=EXPORT_BITS,
                      tolerance_bits=tolerance_bits, panel_eval_limit=panel_eval_limit,
                      entry_eval_limit=entry_eval_limit, total_eval_limit=total_eval_limit,
                      depth_limit=depth_limit, degree_limit=degree_limit)
    payload = dict(schema=SCHEMA, status="UNDETERMINED", eta=_rational_text(ETA),
                   dimension=32, labels=_labels(), parameters=parameters,
                   provenance={"inherited_source": str(SOURCE), "source_sha256": SOURCE_SHA256,
                               "source_head": "d0a4eca2a49a9167848ba4d5d5cfd7eb03d355f7",
                               "python": platform.python_version(), "python_executable": sys.executable,
                               "global_knowledge": "main@4fa7d7d"},
                   diagnostics={"evaluations": 0, "panels": 0, "completed_cross_entries": 0})
    diagnostics = payload["diagnostics"]
    try:
        if (type(precision) is not int or precision < 128 or
                type(cauchy_cutoff) is not int or cauchy_cutoff < 2 or
                any(type(v) is not int or v <= 0 for v in (tolerance_bits, panel_eval_limit,
                    entry_eval_limit, total_eval_limit, depth_limit, degree_limit))):
            raise Undetermined("invalid precision, cutoff, or resource parameters")
        flint, arb, acb, fmpq, inherited = _load_backend(precision)
        payload["provenance"]["python_flint"] = flint.__version__
        old, shell, basis = _basis_groups(inherited)
        diagnostics["denominator_gap_lower"] = _check_tail_domain(basis, K, P, arb)
        A, D, B_prime = _components(old, shell, basis, K, P, precision, inherited,
                                  arb, fmpq, component_output, reuse_components, diagnostics)
        B, entry_diagnostics = [[None] * 16 for _ in range(16)], []
        limits = {key: parameters[key] for key in ("tolerance_bits", "panel_eval_limit",
                  "entry_eval_limit", "total_eval_limit", "depth_limit", "degree_limit")}
        for i, bi in enumerate(old):
            for j, bj in enumerate(shell):
                C, evidence = _cauchy_entry(bi, bj, cauchy_cutoff, inherited, arb, acb,
                                            fmpq, limits, diagnostics)
                B[i][j] = B_prime[i][j] + C
                evidence.update(i=i, j=j, cauchy_complete=export_ball(C))
                entry_diagnostics.append(evidence)
                diagnostics["completed_cross_entries"] += 1
            print(f"Cauchy row {i+1}/16 enclosed; evaluations={diagnostics['evaluations']}", flush=True)
            if progress_output is not None:
                _write_json(progress_output, dict(schema="owner_rh_reference_progress_v1",
                    parameters=parameters, source_sha256=SOURCE_SHA256,
                    completed_cross_entries=diagnostics["completed_cross_entries"],
                    entries=entry_diagnostics))
        eta_squared = arb(fmpq(81, 100))
        H = [[arb(0) for _ in range(32)] for __ in range(32)]
        for i in range(16):
            for j in range(16):
                H[i][j] = eta_squared * A[i][j]
                H[i+16][j+16] = D[i][j]
                H[i][j+16] = B[i][j]
                H[j+16][i] = B[i][j]
        H = _mirror_upper(H)
        bounds = _export_matrix(H)
        radii = [[(Fraction(v["hi"])-Fraction(v["lo"]))/2 for v in row] for row in bounds]
        payload.update(status="ENCLOSED", bounds=bounds,
            error_budget={"all_tails_in_bounds": True, "additional_beta": "0/1",
                          "arch_tail_method": "inherited exact prefix + Hurwitz-zeta powers + remainder balls",
                          "cauchy_tail_per_entry": _rational_text(Fraction(10240, cauchy_cutoff**3)),
                          "row_radius_sum_upper": _rational_text(max(map(sum, radii))),
                          "finite_integration_radii_and_tails": entry_diagnostics},
            target={"basis": "four labelled branches, 8 Dirichlet sine modes each",
                    "diagonals": "complete arch + prime + both pole channels",
                    "cross": "prime + Cauchy; no pole cross or regular arch cross",
                    "matrix": "[[81/100 A, B_ref], [B_ref^T, D]]",
                    "frequency_cutoff": None,
                    "scope": "matrix enclosure only; no inertia count and no RH conclusion"})
    except Exception as error:
        payload.update(status="UNDETERMINED", reason=f"{type(error).__name__}: {error}")
        payload.pop("bounds", None)
    payload["diagnostics"]["elapsed_seconds"] = time.perf_counter() - started
    return payload


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--K", type=int, default=256)
    parser.add_argument("--P", type=int, default=10)
    parser.add_argument("--prec", type=int, default=384)
    parser.add_argument("--cauchy-cutoff", type=int, default=20000)
    parser.add_argument("--tolerance-bits", type=int, default=60)
    parser.add_argument("--panel-eval-limit", type=int, default=4000)
    parser.add_argument("--entry-eval-limit", type=int, default=100000)
    parser.add_argument("--total-eval-limit", type=int, default=2000000)
    parser.add_argument("--depth-limit", type=int, default=20)
    parser.add_argument("--degree-limit", type=int, default=32)
    parser.add_argument("--components-output", type=Path)
    parser.add_argument("--reuse-components", type=Path)
    parser.add_argument("--progress-output", type=Path)
    args = parser.parse_args()
    payload = build_reference_bounds(K=args.K, P=args.P, precision=args.prec,
        cauchy_cutoff=args.cauchy_cutoff, tolerance_bits=args.tolerance_bits,
        panel_eval_limit=args.panel_eval_limit, entry_eval_limit=args.entry_eval_limit,
        total_eval_limit=args.total_eval_limit, depth_limit=args.depth_limit,
        degree_limit=args.degree_limit, component_output=args.components_output,
        reuse_components=args.reuse_components, progress_output=args.progress_output)
    _write_json(args.output, payload)
    print(json.dumps({"status": payload["status"], "output": str(args.output),
                      "reason": payload.get("reason"), "diagnostics": payload["diagnostics"]},
                     separators=(",", ":")), flush=True)
    return 0 if payload["status"] == "ENCLOSED" else 2


if __name__ == "__main__":
    raise SystemExit(main())
